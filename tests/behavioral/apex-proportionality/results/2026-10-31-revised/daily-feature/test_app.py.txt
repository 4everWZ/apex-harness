import unittest
from app import handle
class AppTests(unittest.TestCase):
    def test_live(self):
        self.assertEqual(handle("GET", "/health/live", lambda: False), (200, {"alive": True}))
    def test_missing(self):
        self.assertEqual(handle("POST", "/missing", lambda: True)[0], 404)
    def test_ready(self):
        for readiness, status in ((True, 200), (False, 503)):
            with self.subTest(ready=readiness):
                self.assertEqual(
                    handle("GET", "/health/ready", lambda: readiness),
                    (status, {"ready": readiness}),
                )
    def test_ready_requires_get(self):
        self.assertEqual(
            handle("POST", "/health/ready", lambda: self.fail("unexpected readiness call")),
            (404, {"error": "not found"}),
        )
