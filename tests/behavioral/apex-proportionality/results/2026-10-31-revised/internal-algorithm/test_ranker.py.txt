import unittest
from ranker import choose
class RankerTests(unittest.TestCase):
    def test_order(self):
        items = [{"score": 1, "name": "b"}, {"score": 2, "name": "z"}, {"score": 1, "name": "a"}]
        self.assertEqual(choose(items, 2), [items[1], items[2]])
    def test_empty(self):
        self.assertEqual(choose([], 3), [])

    def test_ties_and_duplicates_preserve_identity_and_input_order(self):
        first = {"score": 2, "name": "a"}
        second = {"score": 2, "name": "a"}
        items = [{"score": 1, "name": "z"}, first, second, first]
        for k in (1, 2, 3, 4, 8):
            with self.subTest(k=k):
                expected = sorted(items, key=lambda row: (-row["score"], row["name"]))[:k]
                actual = choose(items, k)
                self.assertEqual(len(actual), len(expected))
                for selected, original in zip(actual, expected):
                    self.assertIs(selected, original)

    def test_nonpositive_k(self):
        items = [{"score": 1, "name": "a"}]
        for k in (0, -1, -10):
            with self.subTest(k=k):
                self.assertEqual(choose(items, k), [])
